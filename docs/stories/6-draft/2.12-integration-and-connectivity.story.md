### Integration Architecture

**OAuth Implementation:**

```typescript
// Generic OAuth handler
class OAuthManager {
  private tokens: Map<string, OAuthToken> = new Map();

  async authenticate(service: string): Promise<OAuthToken> {
    const config = OAuthConfigs[service];

    // Generate auth URL
    const authUrl = `${config.authUrl}?${new URLSearchParams({
      client_id: config.clientId,
      redirect_uri: config.redirectUri,
      scope: config.scopes.join(' '),
      response_type: 'code',
      state: generateState()
    })}`;

    // Open authentication flow
    const authResult = await this.openAuthFlow(authUrl, config.redirectUri);

    // Exchange code for token
    const token = await this.exchangeCodeForToken(service, authResult.code);
    this.tokens.set(service, token);

    return token;
  }

  async refreshToken(service: string): Promise<OAuthToken> {
    const currentToken = this.tokens.get(service);
    if (!currentToken?.refreshToken) {
      throw new Error('No refresh token available');
    }

    const newToken = await this.refreshAccessToken(service, currentToken.refreshToken);
    this.tokens.set(service, newToken);
    return newToken;
  }
}
```

**Data Synchronization Engine:**

```typescript
interface SyncEngine {
  syncHealthData(): Promise<SyncResult>;
  syncCalendarEvents(): Promise<SyncResult>;
  syncWearableData(): Promise<SyncResult>;
  scheduleSyncs(): void;
  handleSyncConflicts(conflicts: DataConflict[]): Promise<void>;
}

class HealthDataSync implements SyncEngine {
  async syncHealthData(): Promise<SyncResult> {
    const results: SyncResult = { success: [], errors: [] };

    // Apple Health
    if (this.isAppleHealthAvailable()) {
      try {
        const healthData = await this.fetchAppleHealthData();
        const processedData = this.processHealthData(healthData);
        results.success.push({ service: 'apple-health', records: processedData.length });
      } catch (error) {
        results.errors.push({ service: 'apple-health', error });
      }
    }

    // Google Fit
    if (this.isGoogleFitAvailable()) {
      try {
        const fitData = await this.fetchGoogleFitData();
        const processedData = this.processHealthData(fitData);
        results.success.push({ service: 'google-fit', records: processedData.length });
      } catch (error) {
        results.errors.push({ service: 'google-fit', error });
      }
    }

    return results;
  }

  private processHealthData(rawData: any): HabitData[] {
    return rawData.map((record: any) => ({
      source: record.source,
      type: this.mapHealthMetricToHabit(record.type),
      value: record.value,
      timestamp: new Date(record.timestamp),
      metadata: record.metadata
    }));
  }
}
```

```typescript
// React Native Health library integration
import Health from 'react-native-health';

const syncAppleHealth = async () => {
  const permissions = {
    permissions: {
      read: [
        Health.Constants.Permissions.Steps,
        Health.Constants.Permissions.DistanceWalkingRunning,
        Health.Constants.Permissions.FlightsClimbed,
        Health.Constants.Permissions.SleepAnalysis,
        Health.Constants.Permissions.HeartRate,
        Health.Constants.Permissions.ActiveEnergyBurned
      ],
      write: []
    }
  };

  const isAuthorized = await Health.isAvailable(permissions);
  if (!isAuthorized) {
    throw new Error('Apple Health authorization failed');
  }

  // Fetch data for the last 30 days
  const endDate = new Date();
  const startDate = new Date();
  startDate.setDate(endDate.getDate() - 30);

  const steps = await Health.getDailyStepCountSamples(startDate, endDate);
  const workouts = await Health.getWorkouts(startDate, endDate);
  const sleep = await Health.getSleepSamples(startDate, endDate);

  return {
    steps,
    workouts,
    sleep,
    syncDate: new Date()
  };
};
```

**Google Fit Integration:**

```typescript
// Google Fit REST API integration
const syncGoogleFit = async (accessToken: string) => {
  const fitApi = new GoogleFitApi(accessToken);

  const endTime = Date.now();
  const startTime = endTime - (30 * 24 * 60 * 60 * 1000); // 30 days ago

  const dataSources = await fitApi.getDataSources();
  const results = [];

  for (const source of dataSources) {
    if (source.dataType.name.includes('step_count')) {
      const steps = await fitApi.aggregateData({
        dataSourceId: source.dataStreamId,
        startTime,
        endTime,
        aggregationBy: [FitnessAggregation.ByDay]
      });
      results.push(...steps);
    }
  }

  return results;
};
```

```typescript
// Google Calendar API integration
const syncGoogleCalendar = async (accessToken: string) => {
  const calendarApi = new GoogleCalendarApi(accessToken);

  // Get calendar list
  const calendars = await calendarApi.listCalendars();
  const primaryCalendar = calendars.find(cal => cal.primary);

  if (!primaryCalendar) {
    throw new Error('No primary calendar found');
  }

  // Sync habit blocks
  const habitEvents = await createHabitEvents(primaryCalendar.id);
  const results = [];

  for (const event of habitEvents) {
    try {
      const createdEvent = await calendarApi.createEvent(primaryCalendar.id, event);
      results.push({ success: true, eventId: createdEvent.id });
    } catch (error) {
      results.push({ success: false, error });
    }
  }

  return results;
};

const createHabitEvents = (calendarId: string): CalendarEvent[] => {
  return habits.map(habit => ({
    summary: `${habit.emoji} ${habit.name}`,
    description: `Habit tracking: ${habit.description}`,
    start: {
      dateTime: calculateNextHabitTime(habit.reminderTime),
      timeZone: Intl.DateTimeFormat().resolvedOptions().timeZone
    },
    end: {
      dateTime: calculateHabitEndTime(habit.reminderTime, habit.duration),
      timeZone: Intl.DateTimeFormat().resolvedOptions().timeZone
    },
    recurrence: [
      `RRULE:FREQ=DAILY;BYHOUR=${habit.reminderTime.getHours()};BYMINUTE=${habit.reminderTime.getMinutes()}`
    ],
    reminders: {
      useDefault: false,
      overrides: [
        { method: 'popup', minutes: 15 },
        { method: 'notification', minutes: 5 }
      ]
    }
  }));
};
```

### Performance Optimization

**Background Sync Strategy:**

```typescript
const BackgroundSyncManager = {
  scheduleSyncs: () => {
    // Schedule health data sync every 6 hours
    TaskManager.defineTask('health-sync', {
      taskType: 'backgroundFetch',
      minimumInterval: 6 * 60 * 60 * 1000, // 6 hours
      stopOnTerminate: false,
      startOnBoot: true,
    });

    // Schedule calendar sync daily
    TaskManager.defineTask('calendar-sync', {
      taskType: 'backgroundFetch',
      minimumInterval: 24 * 60 * 60 * 1000, // 24 hours
      stopOnTerminate: false,
      startOnBoot: true,
    });
  },

  handleBackgroundTask: async (taskName: string) => {
    try {
      switch (taskName) {
        case 'health-sync':
          await healthDataSync.syncHealthData();
          break;
        case 'calendar-sync':
          await calendarSync.syncCalendarEvents();
          break;
      }
      return BackgroundFetch.BackgroundFetchResult.NewData;
    } catch (error) {
      console.error(`Background sync failed for ${taskName}:`, error);
      return BackgroundFetch.BackgroundFetchResult.Failed;
    }
  }
};
```
