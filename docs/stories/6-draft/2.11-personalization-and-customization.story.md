## User Story

**As a** user,
**I want** extensive personalization and customization options,
**so that** I can tailor the app's appearance, functionality, and experience to match my personal style and preferences.

**Dynamic Theme Implementation:**

```typescript
// Theme provider component
const ThemeProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [currentTheme, setCurrentTheme] = useState<Theme>(defaultTheme);
  const [customThemes, setCustomThemes] = useState<Theme[]>([]);

  const themeContext = {
    currentTheme,
    setTheme: (theme: Theme) => {
      setCurrentTheme(theme);
      saveThemePreference(theme.id);
    },
    createCustomTheme: (themeData: Partial<Theme>) => {
      const newTheme = {
        ...defaultTheme,
        id: generateId(),
        name: themeData.name || 'Custom Theme',
        ...themeData
      };
      setCustomThemes(prev => [...prev, newTheme]);
      saveCustomThemes([...customThemes, newTheme]);
      return newTheme;
    },
    availableThemes: [...builtInThemes, ...customThemes]
  };

  return (
    <ThemeContext.Provider value={themeContext}>
      <View style={{ backgroundColor: currentTheme.colors.background }}>
        {children}
      </View>
    </ThemeContext.Provider>
  );
};
```

```typescript
// Widget container component
const WidgetContainer: React.FC<{ widget: Widget }> = ({ widget }) => {
  const renderWidget = () => {
    switch (widget.type) {
      case 'habit-list':
        return <HabitListWidget config={widget.config as WidgetConfig['habit-list']} />;
      case 'progress-ring':
        return <ProgressRingWidget config={widget.config as WidgetConfig['progress-ring']} />;
      case 'streak-counter':
        return <StreakCounterWidget config={widget.config as WidgetConfig['streak-counter']} />;
      case 'quote':
        return <QuoteWidget config={widget.config as WidgetConfig['quote']} />;
      case 'calendar':
        return <CalendarWidget config={widget.config as WidgetConfig['calendar']} />;
      case 'stats':
        return <StatsWidget config={widget.config as WidgetConfig['stats']} />;
      default:
        return null;
    }
  };

  return (
    <View
      style={[
        styles.widgetContainer,
        {
          width: widget.size.width,
          height: widget.size.height,
          backgroundColor: currentTheme.colors.surface,
          borderRadius: currentTheme.borderRadius.md,
          shadowColor: currentTheme.shadows.medium
        }
      ]}
    >
      {renderWidget()}
    </View>
  );
};
```

### Performance Optimization

**Theme Caching:**

```typescript
const themeCache = new Map<string, Theme>();

const getTheme = (themeId: string): Theme => {
  if (themeCache.has(themeId)) {
    return themeCache.get(themeId)!;
  }

  const theme = loadThemeFromStorage(themeId);
  themeCache.set(themeId, theme);
  return theme;
};
```

### Accessibility Implementation

**Text Scaling Support:**

```typescript
const scaledFontSize = (baseSize: number) => {
  const userScale = getUserTextScale();
  return baseSize * userScale;
};

const ScaledText: React.FC<{ style?: any; children: React.ReactNode }> = ({ style, children }) => {
  const scaledStyle = {
    ...style,
    fontSize: style?.fontSize ? scaledFontSize(style.fontSize) : scaledFontSize(16)
  };

  return <Text style={scaledStyle}>{children}</Text>;
};
```

**Color Blind Support:**

```typescript
const adaptColorsForColorBlindness = (colors: ThemeColors, mode: ColorBlindMode) => {
  const adapters = {
    protanopia: (colors: ThemeColors) => ({
      ...colors,
      primary: adjustHue(colors.primary, 15),
      success: convertToBlue(colors.success)
    }),
    deuteranopia: (colors: ThemeColors) => ({
      ...colors,
      primary: adjustHue(colors.primary, -15),
      error: enhanceBrightness(colors.error)
    }),
    tritanopia: (colors: ThemeColors) => ({
      ...colors,
      accent: adjustSaturation(colors.accent, 0.5)
    })
  };

  return adapters[mode]?.(colors) || colors;
};
```
