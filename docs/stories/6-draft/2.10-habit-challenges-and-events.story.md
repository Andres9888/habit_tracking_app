### Challenge System Architecture

**Challenge Types Implementation:**

1. **Duration-Based Challenges:**
   ```typescript
   const evaluateDurationChallenge = (participation: ChallengeParticipation) => {
     const daysActive = calculateActiveDays(participation.progress);
     const totalDays = participation.challenge.requirements.duration;
     return {
       isCompleted: daysActive >= totalDays,
       progress: (daysActive / totalDays) * 100
     };
   };
   ```

2. **Streak-Based Challenges:**
   ```typescript
   const evaluateStreakChallenge = (participation: ChallengeParticipation) => {
     const currentStreak = participation.progress.currentStreak;
     const targetStreak = participation.challenge.requirements.targetStreak;
     return {
       isCompleted: currentStreak >= targetStreak,
       progress: Math.min((currentStreak / targetStreak) * 100, 100)
     };
   };
   ```

const calculateTeamProgress = (teamId: string) => {
  const members = getTeamMembers(teamId);
  const individualProgress = members.map(member =>
    getChallengeProgress(member.userId, team.challengeId)
  );

  return {
    averageProgress: average(individualProgress.map(p => p.progress)),
    totalCompletions: sum(individualProgress.map(p => p.completions)),
    activeMembers: individualProgress.filter(p => p.isActive).length,
    teamStreak: calculateTeamStreak(individualProgress)
  };
};

```typescript
const distributeRewards = async (participation: ChallengeParticipation) => {
  const rewards = [];

  // Base completion rewards
  if (participation.status === 'completed') {
    rewards.push({
      type: 'xp',
      value: participation.challenge.rewards.xp
    });
  }

  // Performance bonuses
  const leaderboardPosition = getLeaderboardPosition(participation.userId, participation.challengeId);
  if (leaderboardPosition <= 10) {
    rewards.push({
      type: 'badge',
      id: `top_10_${participation.challenge.id}`
    });
  }

  // Team rewards
  if (participation.teamId) {
    const teamPerformance = getTeamPerformance(participation.teamId);
    if (teamPerformance.isTopTeam) {
      rewards.push({
        type: 'customization',
        id: 'team_champion_2024'
      });
    }
  }

  await grantRewards(participation.userId, rewards);
};
```

### Gamification Elements

**Experience and Leveling:**

```typescript
const calculateLevelProgress = (userId: string) => {
  const totalXP = getUserTotalXP(userId);
  const currentLevel = Math.floor(totalXP / 1000); // 1000 XP per level
  const levelProgress = (totalXP % 1000) / 1000;

  return {
    level: currentLevel,
    xp: totalXP,
    progressToNext: levelProgress,
    xpToNext: 1000 - (totalXP % 1000)
  };
};
```
