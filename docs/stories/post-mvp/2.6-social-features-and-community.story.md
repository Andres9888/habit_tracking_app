## User Story

**As a** user,
**I want** optional social features that allow me to connect with friends, share progress, and participate in community challenges,
**so that** I can build accountability, stay motivated through social support, and feel connected to others on similar habit-building journeys.

### Data Model Enhancements

```typescript
// New social tables in Convex schema:

users: defineTable({
  // existing fields...
  username: v.string(),
  displayName: v.string(),
  avatar: v.optional(v.string()),
  bio: v.optional(v.string()),
  location: v.optional(v.string()),
  privacySettings: v.object({
    profileVisibility: v.string(), // public, friends, private
    habitVisibility: v.string(), // public, friends, private, custom
    allowFriendRequests: v.boolean(),
    showInLeaderboards: v.boolean(),
  }),
  socialPreferences: v.object({
    enableSharing: v.boolean(),
    allowEncouragement: v.boolean(),
    showInCommunityFeed: v.boolean(),
  }),
});

friendships: defineTable({
  requesterId: v.id('users'),
  addresseeId: v.id('users'),
  status: v.string(), // pending, accepted, declined, blocked
  createdAt: v.number(),
  updatedAt: v.number(),
}).index('by_user', ['requesterId', 'status']);

sharedProgress: defineTable({
  userId: v.id('users'),
  habitId: v.optional(v.id('habits')),
  type: v.string(), // completion, achievement, streak
  content: v.string(),
  visibility: v.string(), // public, friends, private
  createdAt: v.number(),
  likes: v.array(v.id('users')),
  comments: v.array(v.any()),
});

encouragements: defineTable({
  fromUserId: v.id('users'),
  toUserId: v.id('users'),
  type: v.string(), // kudos, high_five, message
  message: v.optional(v.string()),
  relatedTo: v.optional(v.id('sharedProgress')),
  createdAt: v.number(),
});

communityChallenges: defineTable({
  name: v.string(),
  description: v.string(),
  type: v.string(), // public, private, team
  creatorId: v.id('users'),
  requirements: v.any(),
  startDate: v.number(),
  endDate: v.number(),
  participants: v.array(v.id('users')),
  teams: v.array(v.any()), // team-based challenges
  isPublic: v.boolean(),
  maxParticipants: v.optional(v.number()),
});
```
